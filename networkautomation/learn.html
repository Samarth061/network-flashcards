M1 — Python ipaddress Module

Concept

Python's stdlib ipaddress module provides ip_address(), ip_network(), IPv4Network, IPv4Address, IPv6Network classes. Key attributes: .network_address, .broadcast_address, .num_addresses, .prefixlen, .packed. Containment: addr in network. Overlap: network.overlaps(other). Supernets/subnets: .supernet(), .subnets(prefixlen_diff=1).

Code

import ipaddress

# Create objects
net = ipaddress.ip_network("192.168.1.0/24")
addr = ipaddress.ip_address("192.168.1.100")

print(net.network_address)   # 192.168.1.0
print(net.broadcast_address) # 192.168.1.255
print(net.num_addresses)     # 256
print(net.prefixlen)         # 24

# Containment check
print(addr in net)  # True

# Iterate first 5 hosts
for host in list(net.hosts())[:5]:
    print(host)

# Overlap check
net2 = ipaddress.ip_network("192.168.1.128/25")
print(net.overlaps(net2))  # True

# Supernet
print(net.supernet())      # 192.168.0.0/23

# Subnets (split into two /25s)
for subnet in net.subnets(prefixlen_diff=1):
    print(subnet)

# IPv6
v6 = ipaddress.ip_network("2001:db8::/32")
print(v6.num_addresses)    # 79228162514264337593543950336
Interview Insight "I use ipaddress.ip_network() for all CIDR operations in production — it validates input and raises ValueError on invalid addresses. For strict host-only addresses use strict=False to allow host bits set."

M2 — IP Arithmetic & CIDR (interview core)

Concept

Manual bit operations for CIDR math — converting dotted-decimal to 32-bit int, applying subnet mask, finding if IP belongs to network. This is the most common LiveCode problem type. Key insight: (ip_int & mask_int) == network_int is the CIDR membership test.

Code

def ipv4_to_int(ip: str) -> int:
    """Convert '192.168.1.1' → 32-bit integer."""
    parts = ip.split('.')
    result = 0
    for part in parts:
        result = (result << 8) | int(part)
    return result

def int_to_ipv4(n: int) -> str:
    """Convert 32-bit integer → '192.168.1.1'."""
    return '.'.join(str((n >> (8 * i)) & 0xFF) for i in range(3, -1, -1))

def parse_cidr(cidr: str):
    """Return (network_int, mask_int, prefix_len) for '192.168.1.0/24'."""
    ip_str, prefix_str = cidr.split('/')
    prefix_len = int(prefix_str)
    mask_int = ((1 << 32) - 1) ^ ((1 << (32 - prefix_len)) - 1)
    ip_int = ipv4_to_int(ip_str)
    network_int = ip_int & mask_int
    return network_int, mask_int, prefix_len

def cidr_in_network(ip: str, cidr: str) -> bool:
    """Check if IP belongs to CIDR block."""
    network_int, mask_int, _ = parse_cidr(cidr)
    return (ipv4_to_int(ip) & mask_int) == network_int

def longest_prefix_match(ip: str, routes: list[str]) -> str | None:
    """Find most-specific matching CIDR for IP. O(n) linear scan."""
    best = None
    best_len = -1
    for cidr in routes:
        net_int, mask_int, prefix_len = parse_cidr(cidr)
        if (ipv4_to_int(ip) & mask_int) == net_int:
            if prefix_len > best_len:
                best_len = prefix_len
                best = cidr
    return best

def cidrs_overlap(cidr_a: str, cidr_b: str) -> bool:
    """Return True if two CIDR blocks share any addresses."""
    net_a, mask_a, plen_a = parse_cidr(cidr_a)
    net_b, mask_b, plen_b = parse_cidr(cidr_b)
    # Ranges overlap iff max(start_a, start_b) <= min(end_a, end_b)
    end_a = net_a | (~mask_a & 0xFFFFFFFF)
    end_b = net_b | (~mask_b & 0xFFFFFFFF)
    return max(net_a, net_b) <= min(end_a, end_b)

# --- Examples ---
print(ipv4_to_int("192.168.1.1"))       # 3232235777
print(int_to_ipv4(3232235777))          # 192.168.1.1
print(cidr_in_network("10.0.0.5", "10.0.0.0/24"))   # True
print(longest_prefix_match("10.1.2.3", ["10.0.0.0/8", "10.1.0.0/16", "10.1.2.0/24"]))  # 10.1.2.0/24
print(cidrs_overlap("10.0.0.0/24", "10.0.0.128/25")) # True

cidr_in_network: O(1)  |  longest_prefix_match: O(n) — follow-up: trie reduces to O(prefix_len)

Interview Insight "In interviews I write ipv4_to_int first, then build on it. I mention that production would use a radix trie or the pytricia library for route tables. For IPv6 it's the same logic but 128 bits."

M3 — subprocess + CLI Tools from Python

Concept

subprocess.run() lets Python run system commands (ping, traceroute, dig, nslookup). Key params: capture_output=True (captures stdout/stderr), text=True (returns strings not bytes), timeout=N (raises TimeoutExpired). Check returncode == 0 for success.

Code

import subprocess

def run_ping(host: str, count: int = 4) -> dict:
    """Run system ping and return raw output."""
    try:
        result = subprocess.run(
            ["ping", "-c", str(count), host],
            capture_output=True,
            text=True,
            timeout=30
        )
        return {
            "host": host,
            "returncode": result.returncode,
            "stdout": result.stdout,
            "stderr": result.stderr,
            "success": result.returncode == 0
        }
    except subprocess.TimeoutExpired:
        return {"host": host, "success": False, "error": "timeout"}
    except FileNotFoundError:
        return {"host": host, "success": False, "error": "ping not found"}

def run_traceroute(host: str, max_hops: int = 30) -> dict:
    """Run traceroute."""
    try:
        result = subprocess.run(
            ["traceroute", "-m", str(max_hops), host],
            capture_output=True, text=True, timeout=60
        )
        return {"host": host, "output": result.stdout, "success": result.returncode == 0}
    except subprocess.TimeoutExpired:
        return {"host": host, "success": False, "error": "timeout"}

def run_dig(name: str, record_type: str = "A") -> dict:
    """Run dig DNS lookup."""
    try:
        result = subprocess.run(
            ["dig", "+short", record_type, name],
            capture_output=True, text=True, timeout=10
        )
        records = [line.strip() for line in result.stdout.splitlines() if line.strip()]
        return {"name": name, "type": record_type, "records": records}
    except subprocess.TimeoutExpired:
        return {"name": name, "error": "timeout"}

# Usage
ping_result = run_ping("8.8.8.8", count=2)
print(ping_result["stdout"])
Interview Insight "I always set timeout= because network tools can hang indefinitely. I also catch FileNotFoundError in case the system tool isn't installed, and return structured dicts rather than raw strings so callers can programmatically check .success."

M4 — Regex for Network Parsing

Concept

re.compile() pre-compiles patterns for reuse (faster in loops). Named groups (?P<name>...) make match objects self-documenting. re.MULTILINE makes ^ match start of each line (essential for multi-line CLI output). re.search() finds first match, re.findall() returns all matches.

Code

import re

# Compiled constants — define once, reuse in loops
IP_RE    = re.compile(r'\b(\d{1,3}\.){3}\d{1,3}\b')
CIDR_RE  = re.compile(r'\b(\d{1,3}\.){3}\d{1,3}/\d{1,2}\b')
PERCENT_RE = re.compile(r'(\d+(?:\.\d+)?)\s*%')
RTT_RE   = re.compile(r'(\d+(?:\.\d+)?)\s*ms')

# Named groups for structured extraction
PING_STATS_RE = re.compile(
    r'(?P<tx>\d+) packets transmitted, (?P<rx>\d+) received.*?'
    r'(?P<loss>\d+(?:\.\d+)?)% packet loss',
    re.DOTALL
)

PING_RTT_RE = re.compile(
    r'rtt min/avg/max/mdev = '
    r'(?P<min>\d+\.\d+)/(?P<avg>\d+\.\d+)/(?P<max>\d+\.\d+)/(?P<mdev>\d+\.\d+)'
)

# Extract all IPs from text
text = "Gateway: 10.0.0.1, DNS: 8.8.8.8, 8.8.4.4"
ips = IP_RE.findall(text)          # returns list of last-octet matches (use re.finditer for full matches)
all_ips = [m.group() for m in IP_RE.finditer(text)]
print(all_ips)  # ['10.0.0.1', '8.8.8.8', '8.8.4.4']

# Named group access
ping_out = "3 packets transmitted, 3 received, 0% packet loss"
m = PING_STATS_RE.search(ping_out)
if m:
    print(m.group("tx"), m.group("rx"), m.group("loss"))

# Multiline — match start of each line
routes = """
10.0.0.0/8 via 192.168.1.1
172.16.0.0/12 via 10.0.0.1
"""
for m in re.finditer(r'^(\S+/\d+)\s+via\s+(\S+)', routes, re.MULTILINE):
    print(f"Route {m.group(1)} → {m.group(2)}")
Interview Insight "I always compile regex patterns at module level as constants — it's faster (compiled once) and self-documenting. In interviews I reach for named groups immediately so I can do m.group('loss') instead of m.group(1) and never lose track of which capture is which."

M5 — YAML & JSON for Device Inventories

Concept

YAML is human-readable config format. yaml.safe_load() (never yaml.load() — security risk). JSON for API data. Device inventories have hosts/groups/credentials structure. Always loop with per-device try/except so one bad device doesn't stop the run.

Code

import yaml
import json

# YAML device inventory (typical structure)
INVENTORY_YAML = """
defaults:
  username: admin
  password: secret123
  timeout: 30

devices:
  - host: 192.168.1.1
    device_type: cisco_ios
    name: core-sw-01
  - host: 192.168.1.2
    device_type: arista_eos
    name: core-sw-02
  - host: 192.168.1.3
    device_type: cisco_nxos
    name: dist-sw-01
"""

def load_inventory(yaml_str: str) -> dict:
    """Load device inventory from YAML string."""
    return yaml.safe_load(yaml_str)  # safe_load, never load()

def get_devices(inventory: dict) -> list[dict]:
    """Merge defaults into each device entry."""
    defaults = inventory.get("defaults", {})
    devices = []
    for dev in inventory.get("devices", []):
        merged = {**defaults, **dev}  # device-level overrides defaults
        devices.append(merged)
    return devices

# Load and iterate
inventory = load_inventory(INVENTORY_YAML)
devices = get_devices(inventory)

results = []
for device in devices:
    try:
        # simulate connecting
        result = {"host": device["host"], "status": "ok", "name": device["name"]}
        results.append(result)
    except KeyError as e:
        results.append({"host": device.get("host", "?"), "status": "error", "error": f"missing key {e}"})
    except Exception as e:
        results.append({"host": device.get("host", "?"), "status": "error", "error": str(e)})

# Output as JSON
print(json.dumps(results, indent=2))
Interview Insight "I always use yaml.safe_load() — yaml.load() can execute arbitrary Python via YAML tags and is a security vulnerability. I also merge defaults at load time so connection code doesn't need to handle missing keys."

M6 — Parsing Ping Output → JSON

Concept

Parse the multi-line stdout from system ping into a structured dict. Defensive parsing: check if m is not None before accessing groups. Return None for fields that couldn't be parsed rather than crashing. Handles both Linux and macOS ping format.

Code

import re
from typing import Optional

def parse_ping_output(text: str) -> dict:
    """
    Parse ping stdout into structured dict.

    Returns:
      {
        "transmitted": int,
        "received": int,
        "loss_pct": float,
        "rtt_min_ms": float,
        "rtt_avg_ms": float,
        "rtt_max_ms": float,
        "rtt_mdev_ms": float,   # None on macOS
        "reachable": bool
      }
    """
    result = {
        "transmitted": None,
        "received": None,
        "loss_pct": None,
        "rtt_min_ms": None,
        "rtt_avg_ms": None,
        "rtt_max_ms": None,
        "rtt_mdev_ms": None,
        "reachable": False
    }

    # Linux: "3 packets transmitted, 3 received, 0% packet loss"
    stats_re = re.compile(
        r'(?P<tx>\d+) packets transmitted, (?P<rx>\d+) received'
        r'(?:, \+\d+ errors)?'
        r', (?P<loss>\d+(?:\.\d+)?)% packet loss'
    )
    m = stats_re.search(text)
    if m is not None:
        result["transmitted"] = int(m.group("tx"))
        result["received"] = int(m.group("rx"))
        result["loss_pct"] = float(m.group("loss"))
        result["reachable"] = result["received"] > 0

    # Linux RTT: "rtt min/avg/max/mdev = 0.123/0.456/0.789/0.100 ms"
    rtt_re = re.compile(
        r'rtt min/avg/max/mdev = '
        r'(?P<min>\d+\.\d+)/(?P<avg>\d+\.\d+)/(?P<max>\d+\.\d+)/(?P<mdev>\d+\.\d+)'
    )
    m = rtt_re.search(text)
    if m is not None:
        result["rtt_min_ms"]  = float(m.group("min"))
        result["rtt_avg_ms"]  = float(m.group("avg"))
        result["rtt_max_ms"]  = float(m.group("max"))
        result["rtt_mdev_ms"] = float(m.group("mdev"))

    # macOS RTT: "round-trip min/avg/max/stddev = 0.123/0.456/0.789/0.100 ms"
    if result["rtt_avg_ms"] is None:
        macos_rtt_re = re.compile(
            r'round-trip min/avg/max/stddev = '
            r'(?P<min>\d+\.\d+)/(?P<avg>\d+\.\d+)/(?P<max>\d+\.\d+)/(?P<mdev>\d+\.\d+)'
        )
        m = macos_rtt_re.search(text)
        if m is not None:
            result["rtt_min_ms"] = float(m.group("min"))
            result["rtt_avg_ms"] = float(m.group("avg"))
            result["rtt_max_ms"] = float(m.group("max"))
            result["rtt_mdev_ms"] = float(m.group("mdev"))

    return result

# Test
sample = """
PING 8.8.8.8 (8.8.8.8) 56(84) bytes of data.
64 bytes from 8.8.8.8: icmp_seq=1 ttl=118 time=10.3 ms
3 packets transmitted, 3 received, 0% packet loss, time 2003ms
rtt min/avg/max/mdev = 10.123/10.456/10.789/0.100 ms
"""
import json
print(json.dumps(parse_ping_output(sample), indent=2))
Interview Insight "Every regex match I check if m is not None before accessing groups — one malformed line shouldn't crash the whole parser. I return structured None fields rather than partial dicts so callers can always expect the same keys."

M7 — Parsing Traceroute / MTR Output

Concept

Traceroute output is hop-by-hop. Each line: hop number, hostname/IP (or * * * for unreachable), RTT values. Parse into list of dicts. Find first hop where packet loss exceeds a threshold.

Code

import re
from typing import Optional

def parse_traceroute(text: str) -> list[dict]:
    """
    Parse traceroute output into list of hop dicts.
    Each dict: {hop, ip, hostname, rtts, loss_pct, unreachable}
    """
    hops = []
    # Matches: "  1  router.local (192.168.1.1)  1.234 ms  1.345 ms  1.456 ms"
    hop_re = re.compile(
        r'^\s*(\d+)\s+'                             # hop number
        r'(?:(\S+)\s+\((\d+\.\d+\.\d+\.\d+)\)|(\*))' # hostname (ip) or *
        r'(.*)',                                     # RTT values
        re.MULTILINE
    )
    rtt_re = re.compile(r'(\d+(?:\.\d+)?)\s*ms')

    for m in hop_re.finditer(text):
        hop_num = int(m.group(1))
        unreachable = m.group(4) == '*'
        hostname = m.group(2) if not unreachable else None
        ip = m.group(3) if not unreachable else None
        rtt_str = m.group(5) if not unreachable else ""
        rtts = [float(r) for r in rtt_re.findall(rtt_str)]

        hops.append({
            "hop": hop_num,
            "ip": ip,
            "hostname": hostname,
            "rtts": rtts,
            "unreachable": unreachable,
            "avg_rtt_ms": sum(rtts) / len(rtts) if rtts else None
        })
    return hops

def first_lossy_hop(hops: list[dict], threshold_pct: float = 10.0) -> Optional[dict]:
    """Return first hop where estimated loss exceeds threshold."""
    for hop in hops:
        if hop["unreachable"]:
            return hop
        if hop["rtts"]:
            # Rough loss estimate: fewer than 3 RTTs means some probes were lost
            loss_pct = (3 - len(hop["rtts"])) / 3 * 100
            if loss_pct > threshold_pct:
                return hop
    return None
Interview Insight "Traceroute uses TTL expiry — each hop decrements TTL. * * * means the router didn't send ICMP TTL Exceeded back (firewall, or ICMP rate limiting). I always note this to the interviewer — it doesn't necessarily mean the hop is down."

M8 — Parsing BGP Neighbor Summary

Concept

Cisco IOS show bgp summary output has columns: Neighbor, V, AS, MsgRcvd, MsgSent, TblVer, InQ, OutQ, Up/Down, State/PfxRcd. The State/PfxRcd column is key: if it's a number, the session is Established (and the number is the prefix count). If it's text (Idle/Active/Connect/OpenSent/OpenConfirm), it's NOT established.

Code

import re
from typing import Optional

def parse_bgp_summary(text: str) -> list[dict]:
    """
    Parse 'show bgp summary' output.
    Returns list of neighbor dicts with state/prefix info.
    """
    neighbors = []
    # Skip header lines, parse data rows
    # Data lines start with an IP address
    line_re = re.compile(
        r'^(?P<neighbor>\d+\.\d+\.\d+\.\d+)\s+'   # Neighbor IP
        r'\d+\s+'                                    # V (version)
        r'(?P<as>\d+)\s+'                            # AS
        r'\d+\s+\d+\s+\d+\s+\d+\s+\d+\s+'         # MsgRcvd MsgSent TblVer InQ OutQ
        r'(?P<updown>\S+)\s+'                        # Up/Down
        r'(?P<state>\S+)',                           # State/PfxRcd
        re.MULTILINE
    )

    for m in line_re.finditer(text):
        state_field = m.group("state")
        # If state_field is a number → Established (prefix count)
        established = state_field.isdigit()
        neighbors.append({
            "neighbor": m.group("neighbor"),
            "remote_as": int(m.group("as")),
            "updown": m.group("updown"),
            "state": "Established" if established else state_field,
            "prefix_count": int(state_field) if established else None,
            "established": established
        })
    return neighbors

def non_established_neighbors(text: str) -> list[str]:
    """Return list of neighbor IPs not in Established state."""
    return [n["neighbor"] for n in parse_bgp_summary(text) if not n["established"]]

# Test
bgp_out = """
BGP router identifier 10.0.0.1, local AS number 65001
Neighbor        V    AS MsgRcvd MsgSent   TblVer  InQ OutQ Up/Down  State/PfxRcd
10.1.1.1        4 65002    1234    1200        42    0    0 1d02h          250
10.2.2.2        4 65003       5       3         0    0    0 00:02:15   Active
10.3.3.3        4 65004    9999    9876        42    0    0 5d12h          180
"""
print(non_established_neighbors(bgp_out))  # ['10.2.2.2']
Interview Insight "The key insight is state_field.isdigit() — Cisco encodes Established state as the prefix count number. Text means not Established. I state this assumption clearly in the interview: 'I'll distinguish Established by checking if the last column is numeric.'"

M9 — TextFSM for Structured Parsing

Concept

TextFSM is a state-machine template language that converts unstructured CLI output into a list of dicts. Templates define Value fields and Start state with regex rules. The NTC Templates library (network-to-code/ntc-templates) has 500+ vendor templates. Netmiko integrates it directly with use_textfsm=True.

Code

# TextFSM template (save as show_ip_int_brief.textfsm):
# Value INTERFACE (\S+)
# Value IP_ADDRESS (\S+)
# Value STATUS (up|down|administratively down)
# Value PROTOCOL (up|down)
#
# Start
#   ^${INTERFACE}\s+${IP_ADDRESS}\s+YES\s+\S+\s+${STATUS}\s+${PROTOCOL} -> Record
#   ^. -> Continue

import textfsm
import io

TEMPLATE = """Value INTERFACE (\\S+)
Value IP_ADDRESS (\\S+)
Value STATUS (up|down|administratively down)
Value PROTOCOL (up|down)

Start
  ^${INTERFACE}\\s+${IP_ADDRESS}\\s+YES\\s+\\S+\\s+${STATUS}\\s+${PROTOCOL} -> Record
  ^. -> Continue
"""

CLI_OUTPUT = """
Interface              IP-Address      OK? Method Status                Protocol
GigabitEthernet0/0     192.168.1.1     YES manual up                    up
GigabitEthernet0/1     unassigned      YES unset  administratively down down
Loopback0              10.0.0.1        YES manual up                    up
"""

def parse_with_textfsm(template_str: str, output: str) -> list[dict]:
    """Parse CLI output using TextFSM template, return list of dicts."""
    fsm = textfsm.TextFSM(io.StringIO(template_str))
    rows = fsm.ParseText(output)
    return [dict(zip(fsm.header, row)) for row in rows]

results = parse_with_textfsm(TEMPLATE, CLI_OUTPUT)
for r in results:
    print(r)
# {'INTERFACE': 'GigabitEthernet0/0', 'IP_ADDRESS': '192.168.1.1', 'STATUS': 'up', 'PROTOCOL': 'up'}

M10 — Netmiko (Kirk Byers)

Concept

Netmiko is the de-facto Python SSH library for network devices. ConnectHandler(**device) creates a connection. Key methods: send_command() for show commands (waits for prompt), send_config_set([...]) for config mode, enable() for privileged mode. Always use as context manager for auto-disconnect.

Code

from netmiko import ConnectHandler
from netmiko.exceptions import NetmikoTimeoutException, NetmikoAuthenticationException

# Device dictionary — all keys required
device = {
    "device_type": "cisco_ios",   # cisco_ios, cisco_nxos, arista_eos, juniper_junos, linux
    "host": "192.168.1.1",
    "username": "admin",
    "password": "cisco123",
    "secret": "enable_pass",      # only for cisco_ios enable mode
    "port": 22,                   # optional, default 22
    "timeout": 30,                # connection timeout
}

# Single device — context manager ensures disconnect
def get_interface_status(device: dict) -> list[dict]:
    try:
        with ConnectHandler(**device) as conn:
            conn.enable()  # enter privileged mode

            # show command — returns raw string
            raw = conn.send_command("show ip interface brief")

            # OR structured via TextFSM (NTC Templates)
            structured = conn.send_command(
                "show ip interface brief",
                use_textfsm=True
            )

            # Config change
            config_cmds = [
                "interface GigabitEthernet0/1",
                "description UPLINK-TO-CORE",
                "no shutdown"
            ]
            output = conn.send_config_set(config_cmds)

            return structured  # list of dicts
    except NetmikoTimeoutException:
        return [{"error": f"timeout connecting to {device['host']}"}]
    except NetmikoAuthenticationException:

from napalm import get_driver

def napalm_deploy_config(host: str, username: str, password: str,
                          new_config: str, dry_run: bool = True) -> dict:
    """
    Full NAPALM config deployment workflow:
    1. Connect and get pre-state
    2. Load candidate config
    3. Compare (diff)
    4. Commit with auto-revert or discard
    5. Verify post-state
    """
    driver = get_driver("ios")
    device = driver(host, username, password)

    try:
        device.open()

        # Pre-check
        pre_facts = device.get_facts()
        pre_bgp = device.get_bgp_neighbors()

        # Load candidate (merge = adds to running config)
        device.load_merge_candidate(config=new_config)

        # See what would change
        diff = device.compare_config()
        if not diff:
            print("No changes needed — config already matches desired state")
            device.discard_config()
            return {"status": "no-op", "diff": ""}

        print(f"Config diff:\n{diff}")

        if dry_run:
            device.discard_config()
            return {"status": "dry-run", "diff": diff}

        # Commit with 5-minute auto-revert safety net
        device.commit_config(revert_in=300)  # auto-rolls back in 300s unless confirmed

        # Verify post-state
        post_bgp = device.get_bgp_neighbors()
        bgp_ok = _bgp_neighbors_healthy(post_bgp)

        if bgp_ok:
            device.confirm_commit()  # cancel the auto-revert
            return {"status": "committed", "diff": diff, "verified": True}
        else:
            # BGP unhealthy — let auto-revert run, or manually rollback
            device.rollback()
            return {"status": "rolled-back", "diff": diff, "verified": False}

    finally:
        device.close()

M14 — Concurrency: Multi-Device with ThreadPoolExecutor

Concept

SSH connections are I/O-bound, so ThreadPoolExecutor gives true parallelism. Pattern: submit all tasks, then collect results with as_completed(). Catch per-device exceptions inside the worker function — never let one device kill all futures. Returns structured result dict per device.

Code

from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import Callable, Any
import time

def run_on_device(device: dict, task_fn: Callable) -> dict:
    """Worker: run task on one device, catch all exceptions."""
    host = device.get("host", "unknown")
    start = time.time()
    try:
        output = task_fn(device)
        return {
            "host": host,
            "status": "ok",
            "output": output,
            "duration_s": round(time.time() - start, 2),
            "error": None
        }
    except Exception as e:
        return {
            "host": host,
            "status": "error",
            "output": None,
            "duration_s": round(time.time() - start, 2),
            "error": str(e)
        }

def run_on_all_devices(devices: list[dict], task_fn: Callable,
                        max_workers: int = 10) -> list[dict]:
    """Run task on all devices in parallel. Returns all results."""
    results = []

    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        # Submit all tasks
        future_to_device = {
            executor.submit(run_on_device, device, task_fn): device
            for device in devices
        }

        # Collect as they complete (not submission order)
        for future in as_completed(future_to_device):
            result = future.result()  # run_on_device never raises — always returns dict
            results.append(result)
            print(f"[{result['status'].upper()}] {result['host']} ({result['duration_s']}s)")

    return results

# Summary report
def summarize(results: list[dict]) -> dict:
    ok = [r for r in results if r["status"] == "ok"]
    failed = [r for r in results if r["status"] == "error"]
    return {
        "total": len(results),
        "succeeded": len(ok),
        "failed": len(failed),
        "errors": {r["host"]: r["error"] for r in failed}
    }

M15 — Idempotency & Check-Before-Set

Code

from netmiko import ConnectHandler

def idempotent_ntp_deploy(device: dict, desired_ntp_servers: list[str]) -> dict:
    """
    Idempotently ensure NTP servers are configured.
    Only pushes servers not already in running-config.
    """
    with ConnectHandler(**device) as conn:
        conn.enable()

        # 1. Read current state
        running_config = conn.send_command("show running-config | include ntp server")

        # 2. Find what's missing
        to_add = []
        for server in desired_ntp_servers:
            if f"ntp server {server}" not in running_config:
                to_add.append(f"ntp server {server}")

        # 3. Push only the delta
        if not to_add:
            return {"status": "no-op", "added": [], "reason": "already configured"}

        output = conn.send_config_set(to_add)

        # 4. Verify
        new_config = conn.send_command("show running-config | include ntp server")
        verified = all(f"ntp server {s}" in new_config for s in desired_ntp_servers)

        return {
            "status": "ok" if verified else "error",
            "added": to_add,
            "verified": verified,
            "output": output
        }

M17 — Error Handling, Retries & Partial Failure

Concept

Retry transient failures (network blips, SSH auth delay) with exponential backoff + jitter. Never retry on permanent errors (wrong password, wrong device_type). Collect per-device errors into a structured report. Partial success is success — never fail the whole job because one device failed.

Code

import time
import random
import logging
from netmiko.exceptions import NetmikoAuthenticationException

logger = logging.getLogger(__name__)

def retry_with_backoff(fn, args=(), kwargs={},
                        max_attempts=3, base_wait=1.0,
                        permanent_exceptions=(NetmikoAuthenticationException,)):
    """
    Retry fn with exponential backoff + jitter.
    Immediately re-raises 'permanent' exceptions (no retry).
    """
    for attempt in range(1, max_attempts + 1):
        try:
            return fn(*args, **kwargs)
        except permanent_exceptions:
            raise  # auth failure → no point retrying
        except Exception as e:
            if attempt == max_attempts:
                raise  # exhausted retries
            wait = base_wait * (2 ** (attempt - 1)) * (0.5 + random.random())
            logger.warning(
                "attempt=%d/%d fn=%s error=%s retrying_in=%.2fs",
                attempt, max_attempts, fn.__name__, e, wait
            )
            time.sleep(wait)

def run_with_partial_failure(devices: list[dict], task_fn) -> dict:
    """Run task on all devices, collecting success and failure separately."""
    succeeded = []
    failed = []
    errors = {}

    for device in devices:
        host = device.get("host", "unknown")
        try:
            result = retry_with_backoff(task_fn, args=(device,))
            succeeded.append(host)
        except Exception as e:
            failed.append(host)
            errors[host] = str(e)
            logger.error("host=%s error=%s", host, e)

    return {
        "succeeded": succeeded,
        "failed": failed,
        "errors": errors,
        "total": len(devices),
        "success_rate": len(succeeded) / len(devices) if devices else 0
    }
Interview Insight "I distinguish transient vs permanent errors. Auth failures never benefit from retry — NetmikoAuthenticationException is permanent. Timeout is transient. Jitter (* (0.5 + random.random())) prevents thundering herd when all devices come back online simultaneously."

M19 — Sliding Window

Concept

Two-pointer technique for problems on contiguous sequences. Expand right, shrink left to maintain a constraint. O(n) time. Network context: longest clean window in log stream, max window with packet loss below threshold.

Code

def length_of_longest_substring(s: str) -> int:
    """
    Classic sliding window: longest substring without repeating chars.
    Time: O(n), Space: O(min(n, charset_size))
    """
    char_index = {}   # char → most recent index seen
    left = 0
    max_len = 0

    for right, char in enumerate(s):
        # If char seen and its last position is within our window
        if char in char_index and char_index[char] >= left:
            left = char_index[char] + 1  # shrink window
        char_index[char] = right
        max_len = max(max_len, right - left + 1)

    return max_len

# Network context: longest window of logs with error_rate < threshold
def longest_clean_log_window(logs: list[dict], max_error_rate: float) -> int:
    """Return length of longest contiguous log window with error_rate <= threshold."""
    left = 0
    max_len = 0
    error_count = 0

    for right in range(len(logs)):
        if logs[right]["level"] == "ERROR":
            error_count += 1

        # Shrink if error rate too high
        while right - left + 1 > 0:
            window_size = right - left + 1
            if error_count / window_size <= max_error_rate:
                break
            if logs[left]["level"] == "ERROR":
                error_count -= 1
            left += 1

        max_len = max(max_len, right - left + 1)

    return max_len

# Trace: s = "abcba"
# right=0 (a): window=a, max=1
# right=1 (b): window=ab, max=2
# right=2 (c): window=abc, max=3
# right=3 (b): b seen at 1, left moves to 2. window=cb, max=3
# right=4 (a): a seen at 0 < left=2, ok. window=cba, max=3
print(length_of_longest_substring("abcba"))  # 3

Time: O(n)  |  Space: O(1) for ASCII charset

Interview Insight "Say aloud: 'I'll use a sliding window — two pointers expanding right, shrinking left to maintain the no-duplicate invariant. I track the last-seen index of each character so I can jump left directly instead of stepping.' Time O(n), space O(1) for ASCII."

M20 — HashMap & Top-K

Concept

Count frequencies with a dict, then use heapq.nlargest(k, freq_dict, key=freq_dict.get) for top-K. O(n log k) time. Network context: top talkers by IP, top error types in logs, top BGP prefixes.

import heapq
from collections import Counter

def top_k_frequent(items: list, k: int) -> list:
    """Return k most frequent items. O(n log k)."""
    freq = Counter(items)  # or: freq = {}; for x in items: freq[x] = freq.get(x,0)+1
    return heapq.nlargest(k, freq, key=freq.get)

# Network: top-K source IPs from flow records
def top_k_talkers(flows: list[dict], k: int) -> list[dict]:
    """Return top-K source IPs by byte count."""
    byte_counts = {}
    for flow in flows:
        src = flow["src_ip"]
        byte_counts[src] = byte_counts.get(src, 0) + flow["bytes"]

    top_ips = heapq.nlargest(k, byte_counts, key=byte_counts.get)
    return [{"ip": ip, "bytes": byte_counts[ip]} for ip in top_ips]

# O(n) alternative for exact top-1: max(freq, key=freq.get)
# O(n log n) alternative: sorted(freq, key=freq.get, reverse=True)[:k]

def top_k_log_errors(logs: list[str], k: int) -> list[tuple]:
    """Top-K error types from log lines."""
    freq = Counter(logs)
    return freq.most_common(k)

print(top_k_frequent(["INFO","ERROR","ERROR","WARN","ERROR","INFO"], 2))
# ['ERROR', 'INFO']

M21 — Stack

Concept

LIFO structure for matching/nesting problems. Push open brackets/tokens, pop when you see a close. If stack is empty when you try to pop → mismatch. If stack not empty at end → unclosed. Network context: validate nested config blocks (router bgp/address-family), JSON/YAML structure check.

Code

def is_valid_brackets(s: str) -> bool:
    """
    Valid parentheses/brackets. Time O(n), Space O(n).
    """
    stack = []
    matching = {')': '(', ']': '[', '}': '{'}

    for char in s:
        if char in '([{':
            stack.append(char)
        elif char in ')]}':
            if not stack or stack[-1] != matching[char]:
                return False
            stack.pop()

    return len(stack) == 0

# Network: validate Cisco config block nesting
def validate_config_blocks(config: str) -> list[str]:
    """Check for unclosed config blocks (router, interface, policy-map, etc.)."""
    errors = []
    stack = []
    open_keywords = {"router", "interface", "policy-map", "route-map", "class-map"}

    for line_num, line in enumerate(config.splitlines(), 1):
        stripped = line.strip()
        if not stripped or stripped.startswith('!'):
            continue
        first_word = stripped.split()[0]
        if first_word in open_keywords:
            stack.append((first_word, line_num))
        elif stripped == 'exit' and stack:
            stack.pop()

    for keyword, line_num in stack:
        errors.append(f"Unclosed '{keyword}' block starting at line {line_num}")
    return errors

print(is_valid_brackets("({[]})"))  # True
print(is_valid_brackets("([)]"))    # False
print(is_valid_brackets("{[}"))     # False

M22 — Sort + Merge Intervals

Concept

Sort intervals by start time, then iterate: if current interval overlaps with last merged (curr.start <= last.end), extend end. Otherwise, append new interval. O(n log n) from sort. Network context: merge maintenance windows, aggregate log time spans.

Code

def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge overlapping intervals. Time O(n log n), Space O(n).
    Input: [[1,3],[2,6],[8,10],[15,18]]
    Output: [[1,6],[8,10],[15,18]]
    """
    if not intervals:
        return []

    intervals.sort(key=lambda x: x[0])  # sort by start
    merged = [intervals[0]]

    for curr in intervals[1:]:
        last = merged[-1]
        if curr[0] <= last[1]:         # overlap: curr.start <= last.end
            last[1] = max(last[1], curr[1])  # extend end
        else:
            merged.append(curr)        # no overlap: new interval

    return merged

# Network: merge maintenance windows
def merge_maintenance_windows(windows: list[dict]) -> list[dict]:
    """Merge overlapping scheduled maintenance windows."""
    if not windows:
        return []

    # Convert to [start, end] for sorting, carry description
    sorted_w = sorted(windows, key=lambda w: w["start"])
    merged = [sorted_w[0].copy()]

    for curr in sorted_w[1:]:
        last = merged[-1]
        if curr["start"] <= last["end"]:
            last["end"] = max(last["end"], curr["end"])
            last["description"] += f" + {curr['description']}"
        else:
            merged.append(curr.copy())

    return merged

print(merge_intervals([[1,3],[2,6],[8,10],[15,18]]))
# [[1, 6], [8, 10], [15, 18]]

Time: O(n log n)  |  Space: O(n)

Interview Insight "The key insight is sort first, then one pass. After sorting, you only need to compare each interval to the last merged one (not all previous). The overlap condition is curr.start <= last.end — equal starts or touching counts as overlap."

M23 — BFS / DFS (Network Topology)

Concept

BFS (breadth-first, uses queue/deque) for shortest path in unweighted graphs. DFS (depth-first, uses recursion or stack) for reachability, cycle detection. Network topology = graph: routers as nodes, links as edges. BFS gives minimum hop count.

Code

from collections import deque

def bfs_shortest_path(graph: dict[str, list[str]], start: str, end: str) -> list[str]:
    """
    BFS shortest path in unweighted graph.
    Time O(V+E), Space O(V).
    graph = {"A": ["B","C"], "B": ["D"], ...}
    """
    if start == end:
        return [start]

    queue = deque([(start, [start])])
    visited = {start}

    while queue:
        node, path = queue.popleft()
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                new_path = path + [neighbor]
                if neighbor == end:
                    return new_path
                visited.add(neighbor)
                queue.append((neighbor, new_path))

    return []  # no path found

def dfs_has_cycle(graph: dict[str, list[str]]) -> bool:
    """
    Detect cycle in DIRECTED graph using three-color DFS.
    WHITE (0) = unvisited, GRAY (1) = in current path, BLACK (2) = done.
    Time O(V+E), Space O(V).
    """
    WHITE, GRAY, BLACK = 0, 1, 2
    color = {node: WHITE for node in graph}

    def dfs(node: str) -> bool:
        color[node] = GRAY
        for neighbor in graph.get(node, []):
            if color[neighbor] == GRAY:
                return True   # back edge → cycle!
            if color[neighbor] == WHITE:
                if dfs(neighbor):
                    return True
        color[node] = BLACK
        return False

    for node in graph:
        if color[node] == WHITE:
            if dfs(node):
                return True
    return False

# Network examples
topology = {
    "R1": ["R2", "R3"],
    "R2": ["R4"],
    "R3": ["R4"],
    "R4": ["R5"],
    "R5": []
}
print(bfs_shortest_path(topology, "R1", "R5"))  # ['R1', 'R2', 'R4', 'R5']

routing_deps = {"A": ["B"], "B": ["C"], "C": ["A"]}  # cycle!
print(dfs_has_cycle(routing_deps))  # True

M24 — Heaps (Latency Percentiles, Top-K)

Concept

heapq is a min-heap. For max-heap: negate values. Two-heap running median: keep lower half in max-heap (negated), upper half in min-heap. Rebalance after each insert. Network context: streaming P50/P99 latency, top-K errors.

Code

import heapq

# Running median using two heaps
class RunningMedian:
    """
    O(log n) insert, O(1) query median.
    max_heap: lower half (stored negated for max-heap behavior)
    min_heap: upper half
    """
    def __init__(self):
        self.max_heap = []  # lower half, negated
        self.min_heap = []  # upper half

    def add(self, val: float) -> None:
        # Push to max_heap first
        heapq.heappush(self.max_heap, -val)

        # Ensure max_heap top <= min_heap top
        if self.min_heap and (-self.max_heap[0]) > self.min_heap[0]:
            heapq.heappush(self.min_heap, -heapq.heappop(self.max_heap))

        # Rebalance sizes (max_heap can be at most 1 larger)
        if len(self.max_heap) > len(self.min_heap) + 1:
            heapq.heappush(self.min_heap, -heapq.heappop(self.max_heap))
        elif len(self.min_heap) > len(self.max_heap):
            heapq.heappush(self.max_heap, -heapq.heappop(self.min_heap))

    def median(self) -> float:
        if len(self.max_heap) == len(self.min_heap):
            return (-self.max_heap[0] + self.min_heap[0]) / 2
        return -self.max_heap[0]

# Test with latency stream
rm = RunningMedian()
for latency in [10, 20, 30, 15, 25]:
    rm.add(latency)
    print(f"After adding {latency}: median = {rm.median()}")
# After adding 10: median = 10
# After adding 20: median = 15.0
# After adding 30: median = 20
# After adding 15: median = 17.5
# After adding 25: median = 20

Insert: O(log n)  |  Median query: O(1)  |  Space: O(n)

Interview Insight "Two-heap median is the canonical streaming statistics problem. The invariant is: max_heap has lower half, min_heap has upper half, sizes differ by at most 1. Python's heapq is a min-heap, so I negate values to simulate max-heap."

M25 — LRU Cache (ARP/DNS Cache Simulation)

Concept

LRU = Least Recently Used. On get: move to most-recently-used end. On put: evict least-recently-used if at capacity. collections.OrderedDict gives O(1) move_to_end() and popitem(last=False).

Code

from collections import OrderedDict

class LRUCache:
    """
    O(1) get and put using OrderedDict.
    Network context: ARP cache, DNS cache, BGP RIB cache.
    """
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()  # maintains insertion/access order

    def get(self, key) -> any:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)  # mark as most recently used
        return self.cache[key]

    def put(self, key, value) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # evict least recently used (front)

    def __repr__(self):
        return f"LRUCache({dict(self.cache)})"

# Simulate ARP cache with capacity 3
arp_cache = LRUCache(3)
arp_cache.put("192.168.1.1", "aa:bb:cc:dd:ee:01")
arp_cache.put("192.168.1.2", "aa:bb:cc:dd:ee:02")
arp_cache.put("192.168.1.3", "aa:bb:cc:dd:ee:03")
arp_cache.get("192.168.1.1")  # access → moves to recent end
arp_cache.put("192.168.1.4", "aa:bb:cc:dd:ee:04")  # evicts 192.168.1.2
print(arp_cache)
# LRUCache({'192.168.1.3': ..., '192.168.1.1': ..., '192.168.1.4': ...})

get: O(1)  |  put: O(1)  |  Space: O(capacity)

Interview Insight "OrderedDict is the right tool here — it's a linked list + hashmap under the hood, giving O(1) move_to_end and popitem. I always explain the network context: ARP caches have fixed capacity and evict least recently used entries, exactly like this."

M26 — Cycle Detection in Directed Graph

Concept

In a directed graph, a cycle means you can reach a node from itself. Standard BFS/DFS with a visited set isn't enough — it can't distinguish between a node in the current search path (back edge = cycle) and a node visited in a previous path (cross edge = no cycle). Three-color DFS solves this: GRAY = currently on stack.

Code

def has_cycle_directed(graph: dict[str, list[str]]) -> bool:
    """
    Three-color DFS cycle detection for directed graphs.
    WHITE = unvisited, GRAY = on current DFS stack, BLACK = fully processed.
    Time O(V+E), Space O(V).
    """
    WHITE, GRAY, BLACK = 0, 1, 2
    color = {}

    def dfs(node: str) -> bool:
        color[node] = GRAY
        for neighbor in graph.get(node, []):
            if neighbor not in color:
                color[neighbor] = WHITE
            if color[neighbor] == GRAY:
                return True  # back edge → cycle
            if color[neighbor] == WHITE:
                if dfs(neighbor):
                    return True
        color[node] = BLACK
        return False

    for node in graph:
        if color.get(node, WHITE) == WHITE:
            if dfs(node):
                return True
    return False

# Why two-color fails for directed graphs:
# A → B → C (no cycle)
# A → C (direct edge)
# When processing A→B→C, C is marked visited.
# When processing A→C directly, C is visited but NOT a cycle.
# Two-color would incorrectly report a cycle.
# Three-color: C is BLACK (already finished), not GRAY (on current path) → no cycle.

# Test cases
print(has_cycle_directed({"A": ["B"], "B": ["C"], "C": []}))    # False
print(has_cycle_directed({"A": ["B"], "B": ["C"], "C": ["A"]})) # True — routing loop!

Time: O(V+E)  |  Space: O(V)

Interview Insight "This comes up for routing loop detection and dependency resolution. I always explain WHY three colors: GRAY means 'on the current recursion stack' — a back edge to GRAY means we can reach the ancestor from a descendant = cycle. BLACK means 'fully explored' — a cross edge to BLACK is not a cycle."
