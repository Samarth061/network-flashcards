1. Longest Substring Without Repeating Characters
Medium

Given a string s, find the length of the longest substring without repeating characters.

Input: s = "abcabcbb" Output: 3 (substring "abc")
Clarify First
  • What should we return for an empty string?
  • Are Unicode characters possible, or ASCII only?
  • Should we return the substring itself or just the length?
  • Is the input always a string (not None)?
Hint

Use a sliding window with two pointers (left and right). Maintain a hash map of each character's last-seen index. When a duplicate is found inside the window, advance left past it. Track the maximum window size.

Solution + Complexity 
def length_of_longest_substring(s: str) -> int:
    char_index = {}
    left = 0
    max_len = 0
    for right, char in enumerate(s):
        if char in char_index and char_index[char] >= left:
            left = char_index[char] + 1
        char_index[char] = right
        max_len = max(max_len, right - left + 1)
    return max_len

# Tests
print(length_of_longest_substring("abcabcbb"))  # 3
print(length_of_longest_substring("bbbbb"))     # 1
print(length_of_longest_substring("pwwkew"))    # 3
print(length_of_longest_substring(""))          # 0
DimensionComplexity
TimeO(n)
SpaceO(min(n, charset))
2. Top-K Most Frequent Log Severities
Medium

Given a list of log severity strings and an integer k, return the k most frequent severities.

Input: logs = ["INFO","ERROR","ERROR","WARN","ERROR","INFO"], k = 2 Output: ["ERROR", "INFO"]
Clarify First
  • How should ties be broken — any order acceptable?
  • Does the order of the result matter?
  • What if k is larger than the number of distinct severities?
  • Can the list be empty?
Hint

Use Counter for frequencies, then heapq.nlargest(k, freq, key=freq.get). Alternatively, Counter.most_common(k) does both steps in one call.

Solution + Complexity 
import heapq
from collections import Counter

def top_k_severities(logs: list[str], k: int) -> list[str]:
    freq = Counter(logs)
    return heapq.nlargest(k, freq, key=freq.get)

# Alternative using most_common
def top_k_severities_v2(logs: list[str], k: int) -> list[str]:
    return [item for item, _ in Counter(logs).most_common(k)]

print(top_k_severities(["INFO","ERROR","ERROR","WARN","ERROR","INFO"], 2))
# ['ERROR', 'INFO']
DimensionComplexity
TimeO(n log k)
SpaceO(n)
3. Merge Overlapping Maintenance Windows
Medium

Given a list of [start, end] time intervals (in minutes since midnight), merge all overlapping intervals and return the merged list.

Input: [[60,120],[90,180],[300,360],[330,420]] Output: [[60,180],[300,420]]
Clarify First
  • Are intervals already sorted, or do we need to sort them?
  • Can start equal end (zero-length interval)?
  • Should adjacent but touching intervals like [1,3] and [3,5] be merged?
  • What if the input list is empty?
Hint

Sort by start time. Iterate keeping a merged list. For each interval, check if its start is ≤ the last merged end — if so, extend; otherwise append as new.

Solution + Complexity 
def merge_windows(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    intervals.sort(key=lambda x: x[0])
    merged = [intervals[0]]
    for curr in intervals[1:]:
        last = merged[-1]
        if curr[0] <= last[1]:
            last[1] = max(last[1], curr[1])
        else:
            merged.append(curr)
    return merged

print(merge_windows([[60,120],[90,180],[300,360],[330,420]]))
# [[60, 180], [300, 420]]
print(merge_windows([[1,4],[4,5]]))  # [[1, 5]] (touching intervals merged)
DimensionComplexity
TimeO(n log n)
SpaceO(n)
4. Valid Parentheses
Easy

Given a string containing only (, ), {, }, [, ], determine if the input is valid (brackets close in correct order).

Input: s = "({[]})" -> True Input: s = "([)]" -> False
Clarify First
  • Is an empty string considered valid?
  • Are only these 6 characters guaranteed in the input?
  • Is there an upper bound on string length?
Hint

Stack approach: push open brackets onto the stack. On a close bracket, check if the top of the stack is the matching open bracket and pop it. At the end, the stack must be empty.

Solution + Complexity 
def is_valid(s: str) -> bool:
    stack = []
    matching = {')': '(', ']': '[', '}': '{'}
    for char in s:
        if char in '([{':
            stack.append(char)
        elif char in ')]}':
            if not stack or stack[-1] != matching[char]:
                return False
            stack.pop()
    return len(stack) == 0

print(is_valid("({[]})"))  # True
print(is_valid("([)]"))    # False
print(is_valid(""))        # True
DimensionComplexity
TimeO(n)
SpaceO(n)
5. BFS Shortest Path in Network Topology
Medium

Given a network topology as an adjacency list and two router names, find the minimum number of hops between them. Return -1 if unreachable.

graph = {"R1":["R2","R3"],"R2":["R4"],"R3":["R4"],"R4":["R5"],"R5":[]} min_hops(graph, "R1", "R5") -> 3
Clarify First
  • Is the graph directed or undirected?
  • Are all edge weights equal to 1 hop?
  • What should we return if start == end?
  • Can a node have no outgoing edges?
Hint

BFS with a deque. Track a visited set to avoid cycles. Store (node, hop_count) tuples in the queue. When you dequeue the destination node, return the hop count.

Solution + Complexity 
from collections import deque

def min_hops(graph: dict[str, list[str]], start: str, end: str) -> int:
    if start == end:
        return 0
    queue = deque([(start, 0)])
    visited = {start}
    while queue:
        node, hops = queue.popleft()
        for neighbor in graph.get(node, []):
            if neighbor == end:
                return hops + 1
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, hops + 1))
    return -1

topology = {"R1":["R2","R3"],"R2":["R4"],"R3":["R4"],"R4":["R5"],"R5":[]}
print(min_hops(topology, "R1", "R5"))  # 3
print(min_hops(topology, "R1", "R1"))  # 0
print(min_hops(topology, "R5", "R1"))  # -1 (directed graph)
DimensionComplexity
TimeO(V + E)
SpaceO(V)
6. Parse Key=Value Config String
Medium

Parse a config string like "host=10.0.0.1 port=22 desc='core switch' enabled=true" into a dict. Values can be quoted strings containing spaces. Unquoted values are plain strings.

Input: "host=10.0.0.1 port=22 desc='core switch'" Output: {"host": "10.0.0.1", "port": "22", "desc": "core switch"}
Clarify First
  • Should double quotes also be supported?
  • Are escaped characters inside quotes possible (e.g., \' or \")?
  • What happens with keys that have no value (no = sign)?
  • Should we strip whitespace from values?
Hint

Use a single regex with alternation: match key= followed by either a double-quoted group, a single-quoted group, or an unquoted run of non-whitespace characters. Use finditer to iterate all matches.

Solution + Complexity 
import re

def parse_kv_config(config: str) -> dict:
    """Parse key=value config string, supporting quoted values."""
    result = {}
    pattern = re.compile(
        r'(\w+)='           # key
        r'(?:'
        r'"([^"]*)"'        # double-quoted value
        r"|'([^']*)'"       # single-quoted value
        r'|(\S+)'           # unquoted value
        r')'
    )
    for m in pattern.finditer(config):
        key = m.group(1)
        value = m.group(2) or m.group(3) or m.group(4) or ""
        result[key] = value
    return result

s = "host=10.0.0.1 port=22 desc='core switch' enabled=true"
print(parse_kv_config(s))
# {'host': '10.0.0.1', 'port': '22', 'desc': 'core switch', 'enabled': 'true'}
DimensionComplexity
TimeO(n)
SpaceO(k) — k = number of keys
7. LRU Cache
Medium

Design a data structure with get(key) and put(key, value) that operates in O(1) time and evicts the least recently used entry when capacity is exceeded.

capacity=2 -> put(1,"a"), put(2,"b"), get(1)->"a" put(3,"c") evicts key 2 get(2) -> -1 (evicted) get(3) -> "c"
Clarify First
  • What should get() return on a cache miss?
  • Is capacity guaranteed to be >= 1?
  • Does put() on an existing key update the value AND recency?
Hint

OrderedDict from collections: move_to_end(key) marks a key as most recently used in O(1), and popitem(last=False) evicts the least recently used in O(1).

Solution + Complexity 
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()

    def get(self, key) -> any:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)

# Test
cache = LRUCache(2)
cache.put(1, "a"); cache.put(2, "b")
print(cache.get(1))  # "a"
cache.put(3, "c")
print(cache.get(2))  # -1  (evicted)
print(cache.get(3))  # "c"
OperationComplexity
get / putO(1)
SpaceO(capacity)
8. Detect Cycle in Directed Graph
Medium

Given a directed graph as an adjacency list, return True if the graph contains a cycle.

{"A":["B"],"B":["C"],"C":["A"]} -> True {"A":["B"],"B":["C"],"C":[]} -> False
Clarify First
  • Can the graph have isolated nodes?
  • Does a self-loop (A -> A) count as a cycle?
  • Are all nodes guaranteed to appear as keys in the adjacency list?
Hint

Three-color DFS: WHITE = unvisited, GRAY = currently on the DFS stack, BLACK = fully processed. If during DFS you reach a GRAY node, you've found a back edge — that's a cycle.

Solution + Complexity 
def has_cycle(graph: dict[str, list[str]]) -> bool:
    WHITE, GRAY, BLACK = 0, 1, 2
    color = {node: WHITE for node in graph}

    def dfs(node: str) -> bool:
        color[node] = GRAY
        for neighbor in graph.get(node, []):
            if neighbor not in color:
                color[neighbor] = WHITE
            if color[neighbor] == GRAY:
                return True
            if color[neighbor] == WHITE and dfs(neighbor):
                return True
        color[node] = BLACK
        return False

    return any(dfs(n) for n in graph if color[n] == WHITE)

print(has_cycle({"A":["B"],"B":["C"],"C":["A"]}))  # True
print(has_cycle({"A":["B"],"B":["C"],"C":[]}))      # False
print(has_cycle({"A":["A"]}))                        # True (self-loop)
DimensionComplexity
TimeO(V + E)
SpaceO(V)
9. Running Median of Latency Stream
Medium

Design a data structure that accepts a stream of latency values (floats) and returns the current median after each insertion.

Stream: [10, 20, 30, 15] Medians: [10, 15.0, 20, 17.5]
Clarify First
  • Are values integers or can they be floats?
  • Do we need all historical medians or just the current one?
  • What if only one element has been inserted so far?
Hint

Two heaps: a max-heap for the lower half (negate values for Python's min-heap) and a min-heap for the upper half. After each insertion, rebalance so sizes differ by at most 1. Median = top of the larger heap, or average of both tops.

Solution + Complexity 
import heapq

class RunningMedian:
    def __init__(self):
        self.max_heap = []  # lower half, values negated
        self.min_heap = []  # upper half

    def add(self, val: float) -> float:
        heapq.heappush(self.max_heap, -val)
        # Ensure ordering: max of lower half <= min of upper half
        if self.min_heap and (-self.max_heap[0]) > self.min_heap[0]:
            heapq.heappush(self.min_heap, -heapq.heappop(self.max_heap))
        # Rebalance sizes
        if len(self.max_heap) > len(self.min_heap) + 1:
            heapq.heappush(self.min_heap, -heapq.heappop(self.max_heap))
        elif len(self.min_heap) > len(self.max_heap):
            heapq.heappush(self.max_heap, -heapq.heappop(self.min_heap))
        return self.median()

    def median(self) -> float:
        if len(self.max_heap) == len(self.min_heap):
            return (-self.max_heap[0] + self.min_heap[0]) / 2
        return float(-self.max_heap[0])

rm = RunningMedian()
for v in [10, 20, 30, 15]:
    print(rm.add(v))  # 10.0, 15.0, 20.0, 17.5
OperationComplexity
add (insert)O(log n)
median queryO(1)
SpaceO(n)
10. Longest Common Prefix of Hostnames
Easy

Given a list of hostnames, find the longest common prefix string. Return "" if none exists.

["core-sw-01","core-sw-02","core-sw-03"] -> "core-sw-0" ["abc","abcdef","ab"] -> "ab" ["dog","racecar","car"] -> ""
Clarify First
  • What if the list is empty?
  • What if there is only one element?
  • Is comparison case-sensitive?
Hint

Sort the list lexicographically. The first and last elements are the most different — find where they diverge character by character. That point is also where the full list diverges.

Solution + Complexity 
def longest_common_prefix(names: list[str]) -> str:
    if not names:
        return ""
    names_sorted = sorted(names)
    first, last = names_sorted[0], names_sorted[-1]
    i = 0
    while i < len(first) and i < len(last) and first[i] == last[i]:
        i += 1
    return first[:i]

print(longest_common_prefix(["core-sw-01","core-sw-02","core-sw-03"]))  # "core-sw-0"
print(longest_common_prefix(["abc","abcdef","ab"]))  # "ab"
print(longest_common_prefix(["dog","racecar","car"]))  # ""
print(longest_common_prefix(["only"]))  # "only"
DimensionComplexity
TimeO(n log n + L) — L = prefix length
SpaceO(1)
11. Two Sum (Port Pair)
Easy

Given a list of port numbers and a target sum, return the indices of the two ports that add up to the target.

ports = [80, 443, 22, 8080], target = 523 Output: [0, 1] (80 + 443 = 523)
Clarify First
  • Is exactly one solution guaranteed to exist?
  • Can the same element be used twice?
  • Is any valid pair acceptable if multiple solutions exist?
  • What should we return if no pair is found?
Hint

Hash map approach: as you iterate, compute complement = target - current and check if it's already in seen. Store seen[value] = index as you go. One pass, O(n).

Solution + Complexity 
def two_sum(nums: list[int], target: int) -> list[int]:
    seen = {}  # value -> index
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
    return []

print(two_sum([80, 443, 22, 8080], 523))  # [0, 1]  (80+443)
print(two_sum([2, 7, 11, 15], 9))         # [0, 1]
print(two_sum([3, 2, 4], 6))              # [1, 2]
DimensionComplexity
TimeO(n)
SpaceO(n)
12. Group Hostnames by Anagram Key
Medium

Given a list of hostnames (or strings), group those that are anagrams of each other (contain the same characters in any order).

["eat","tea","tan","ate","nat","bat"] -> [["eat","tea","ate"], ["tan","nat"], ["bat"]]
Clarify First
  • Is the comparison case-sensitive?
  • What if there are empty strings in the input?
  • Does the order within groups or between groups matter?
Hint

For each string, create a canonical form by sorting its characters: tuple(sorted(s)). Strings with the same canonical form are anagrams. Use a defaultdict(list) keyed by this tuple.

Solution + Complexity 
from collections import defaultdict

def group_anagrams(strs: list[str]) -> list[list[str]]:
    groups = defaultdict(list)
    for s in strs:
        key = tuple(sorted(s))  # canonical form
        groups[key].append(s)
    return list(groups.values())

print(group_anagrams(["eat","tea","tan","ate","nat","bat"]))
# [['eat', 'tea', 'ate'], ['tan', 'nat'], ['bat']]
print(group_anagrams([""]))   # [['']]
print(group_anagrams(["a"]))  # [['a']]

SpaceO(n × L)
NA-1. IPv4 String ↔ 32-bit Integer
Easy

Implement ipv4_to_int(ip) that converts "192.168.1.1" to its 32-bit integer, and int_to_ipv4(n) that converts back.

ipv4_to_int("192.168.1.1") -> 3232235777 int_to_ipv4(3232235777) -> "192.168.1.1" ipv4_to_int("10.0.0.1") -> 167772161
Clarify First
  • Should we validate the IP format, or assume well-formed input?
  • IPv4 only, or should we handle IPv6?
  • Should int_to_ipv4 raise on negative or out-of-range integers?
Hint

Each octet is 8 bits. Build the integer with result = (result << 8) | int(octet) for each octet left-to-right. Reverse: extract each byte with (n >> (8*i)) & 0xFF for i in range(3, -1, -1).

Solution + Complexity 
def ipv4_to_int(ip: str) -> int:
    parts = ip.split('.')
    if len(parts) != 4:
        raise ValueError(f"invalid IP: {ip}")
    result = 0
    for part in parts:
        result = (result << 8) | int(part)
    return result

def int_to_ipv4(n: int) -> str:
    if not (0 <= n <= 0xFFFFFFFF):
        raise ValueError(f"out of range: {n}")
    return '.'.join(str((n >> (8 * i)) & 0xFF) for i in range(3, -1, -1))

print(ipv4_to_int("192.168.1.1"))   # 3232235777
print(int_to_ipv4(3232235777))      # 192.168.1.1
print(ipv4_to_int("10.0.0.1"))      # 167772161
print(int_to_ipv4(0))               # 0.0.0.0
print(int_to_ipv4(0xFFFFFFFF))      # 255.255.255.255
DimensionComplexity
TimeO(1) — always 4 octets
SpaceO(1)
NA-2. CIDR Membership Check
Easy

Write cidr_contains(ip, cidr) that returns True if the given IP address belongs to the CIDR block.

cidr_contains("192.168.1.100", "192.168.1.0/24") -> True cidr_contains("192.168.2.1", "192.168.1.0/24") -> False cidr_contains("10.0.0.1", "10.0.0.0/8") -> True
Clarify First
  • IPv4 only?
  • Should we handle CIDRs with host bits set (e.g., 192.168.1.100/24)?
  • What should /32 and /0 return?
Hint

Build a 32-bit mask from the prefix length: mask = 0xFFFFFFFF ^ ((1 << (32 - plen)) - 1). Then check: (ip_int & mask) == (network_int & mask).

Solution + Complexity 
def cidr_contains(ip: str, cidr: str) -> bool:
    ip_str, prefix_str = cidr.split('/')
    prefix_len = int(prefix_str)
    mask = (
        ((1 << 32) - 1) ^ ((1 << (32 - prefix_len)) - 1)
        if prefix_len < 32 else 0xFFFFFFFF
    )

    def to_int(s):
        result = 0
        for part in s.split('.'):
            result = (result << 8) | int(part)
        return result

    return (to_int(ip) & mask) == (to_int(ip_str) & mask)

print(cidr_contains("192.168.2.1",   "192.168.1.0/24"))  # False
print(cidr_contains("10.0.0.1",      "10.0.0.0/8"))      # True
print(cidr_contains("10.0.0.1",      "10.0.0.1/32"))     # True
DimensionComplexity
TimeO(1)
SpaceO(1)
NA-3. Longest Prefix Match
Medium

Given an IP address and a list of CIDR routes, return the most specific (longest prefix) route that contains the IP. Return None if no route matches.

ip = "10.1.2.3" routes = ["10.0.0.0/8","10.1.0.0/16","10.1.2.0/24","192.168.0.0/16"] -> "10.1.2.0/24"
Clarify First
  • If two routes have the same prefix length, which should be returned?
  • Are all CIDRs guaranteed valid and well-formed?
  • Should a default route (0.0.0.0/0) be treated as a valid match?
  • What if routes list is empty?
Hint

Linear scan: for each route check membership and compare prefix length to best_len. Track the best matching route. Follow-up: a Patricia trie reduces lookup to O(prefix_length) for large route tables.

Solution + Complexity 
def longest_prefix_match(ip: str, routes: list[str]) -> str | None:
    def to_int(s):
        r = 0
        for p in s.split('.'): r = (r << 8) | int(p)
        return r

    def parse_cidr(cidr):
        ip_str, plen_str = cidr.split('/')
        plen = int(plen_str)
        mask = (
            ((1 << 32) - 1) ^ ((1 << (32 - plen)) - 1)
            if plen < 32 else 0xFFFFFFFF
        )
        return to_int(ip_str) & mask, mask, plen

    ip_int = to_int(ip)
    best, best_len = None, -1

    for route in routes:
        try:
            net_int, mask, plen = parse_cidr(route)
            if (ip_int & mask) == net_int and plen > best_len:
                best, best_len = route, plen
        except (ValueError, IndexError):
            continue

    return best

routes = ["10.0.0.0/8","10.1.0.0/16","10.1.2.0/24","192.168.0.0/16"]
print(longest_prefix_match("10.1.2.3", routes))    # 10.1.2.0/24
print(longest_prefix_match("10.5.0.1", routes))    # 10.0.0.0/8
print(longest_prefix_match("172.16.0.1", routes))  # None
DimensionComplexity
TimeO(n) — one pass over routes
SpaceO(1)
Follow-up (Trie)O(prefix_length) lookup
NA-4. Detect CIDR Overlap
Medium

Given two CIDR blocks, return True if they share any IP addresses (including containment).

"10.0.0.0/24" and "10.0.0.128/25" -> True (overlap) "10.0.0.0/25" and "10.0.0.128/25" -> False (adjacent, no overlap) "10.0.0.0/8" and "10.5.5.0/24" -> True (containment)
Clarify First
  • Does containment (one block fully inside the other) count as overlap?
  • Should adjacent CIDRs that share no addresses return False?
  • Can one CIDR be /0 (entire IPv4 space)?
Hint

Convert each CIDR to an integer range [start, end]. Two ranges A and B overlap iff max(start_A, start_B) ≤ min(end_A, end_B).

Solution + Complexity 
def cidrs_overlap(cidr_a: str, cidr_b: str) -> bool:
    def cidr_range(cidr):
        ip_str, plen_str = cidr.split('/')
        plen = int(plen_str)
        mask = (
            ((1 << 32) - 1) ^ ((1 << (32 - plen)) - 1)
            if plen < 32 else 0xFFFFFFFF
        )
        r = 0
        for p in ip_str.split('.'): r = (r << 8) | int(p)
        start = r & mask
        end = start | (~mask & 0xFFFFFFFF)
        return start, end

    s_a, e_a = cidr_range(cidr_a)
    s_b, e_b = cidr_range(cidr_b)
    return max(s_a, s_b) <= min(e_a, e_b)

print(cidrs_overlap("10.0.0.0/24", "10.0.0.128/25"))  # True
print(cidrs_overlap("10.0.0.0/25", "10.0.0.128/25"))  # False (adjacent)
print(cidrs_overlap("10.0.0.0/8",  "10.5.5.0/24"))    # True (containment)
print(cidrs_overlap("192.168.0.0/16", "10.0.0.0/8"))  # False
DimensionComplexity
TimeO(1)
SpaceO(1)

import re, json

def parse_ping_output(text: str) -> dict:
    result = {
        "transmitted": None, "received": None,
        "loss_pct": None, "avg_rtt_ms": None, "reachable": False
    }

    stats_re = re.compile(
        r'(?P<tx>\d+) packets transmitted, (?P<rx>\d+) received'
        r'.*?(?P<loss>\d+(?:\.\d+)?)% packet loss'
    )
    m = stats_re.search(text)
    if m:
        result["transmitted"] = int(m.group("tx"))
        result["received"]    = int(m.group("rx"))
        result["loss_pct"]    = float(m.group("loss"))
        result["reachable"]   = result["received"] > 0

    # Linux RTT line: rtt min/avg/max/mdev = 10.1/10.5/11.0/0.3 ms
    rtt_re = re.compile(r'rtt min/avg/max/mdev = \S+/(?P<avg>\d+\.\d+)/')
    m = rtt_re.search(text)
    if m:
        result["avg_rtt_ms"] = float(m.group("avg"))

    # macOS RTT line: round-trip min/avg/max/stddev = ...
    if result["avg_rtt_ms"] is None:
        mac_re = re.compile(
            r'round-trip min/avg/max/stddev = \S+/(?P<avg>\d+\.\d+)/')
        m = mac_re.search(text)
        if m:
            result["avg_rtt_ms"] = float(m.group("avg"))

    return result

import re
from typing import Optional

def parse_traceroute_first_lossy(text: str) -> Optional[dict]:
    """Return first hop where >50% probes timed out."""
    hop_re = re.compile(r'^\s*(\d+)\s+(.*)', re.MULTILINE)

    for m in hop_re.finditer(text):
        hop_num = int(m.group(1))
        rest = m.group(2).strip()

        probes    = rest.split()
        star_count = probes.count('*')
        rtt_count  = len(re.findall(r'\d+\.?\d*\s*ms', rest))
        total      = star_count + rtt_count

        if total > 0 and star_count / total > 0.5:
            ip_match = re.search(r'\((\d+\.\d+\.\d+\.\d+)\)', rest)
            return {
                "hop": hop_num,
                "ip": ip_match.group(1) if ip_match else None,
                "star_count": star_count,
                "total_probes": total,
                "unreachable": star_count == total
            }
    return None

import json

def check_compliance(running_config: str, required_stanzas: list[str]) -> dict:
    """Check config compliance: present vs missing stanzas."""
    present, missing = [], []
    for stanza in required_stanzas:
        (present if stanza in running_config else missing).append(stanza)

        "present":        present,
        "missing":        missing,
        "total_checked":  total
    }

Step 3 — Code
def ipv4_to_int(ip: str) -> int:
    result = 0
    for part in ip.split('.'):
        result = (result << 8) | int(part)
    return result

def longest_prefix_match(ip: str, routes: list[str]) -> str | None:
    ip_int = ipv4_to_int(ip)
    best, best_len = None, -1

    for route in routes:
        net_str, plen_str = route.split('/')
        plen = int(plen_str)
        mask = ((1 << 32) - 1) ^ ((1 << (32 - plen)) - 1)
        net_int = ipv4_to_int(net_str) & mask

        if (ip_int & mask) == net_int and plen > best_len:
            best, best_len = route, plen

    return best

import re, json
from netmiko import ConnectHandler

def validate_deployment_health(device: dict, test_ip: str) -> dict:
    """
    Post-deployment health check:
    1. Connect and get BGP neighbor summary
    2. Check all BGP neighbors are Established
    3. Ping test_ip from device
    4. Return structured JSON report
    """
    report = {
        "device":               device["host"],
        "test_ip":              test_ip,
        "bgp_ok":               False,
        "bgp_non_established":  [],
        "ping_ok":              False,
        "ping_loss_pct":        None,
        "overall_healthy":      False,
        "errors":               []
    }

    try:
        with ConnectHandler(**device) as conn:
            conn.enable()

            # BGP check
            bgp_raw  = conn.send_command("show bgp summary")
            non_est  = parse_non_established_bgp(bgp_raw)
            report["bgp_non_established"] = non_est
            report["bgp_ok"] = len(non_est) == 0

            # Ping check (from device)
            ping_raw    = conn.send_command(f"ping {test_ip} repeat 5")
            ping_result = parse_cisco_ping(ping_raw)
            report["ping_ok"]       = ping_result.get("reachable", False)
            report["ping_loss_pct"] = ping_result.get("loss_pct")

    except Exception as e:
        report["errors"].append(str(e))

    report["overall_healthy"] = (
        report["bgp_ok"] and report["ping_ok"] and not report["errors"]
    )
    return report


def parse_non_established_bgp(text: str) -> list[str]:
    non_est = []
    for m in re.finditer(
        r'^(\d+\.\d+\.\d+\.\d+)(?:\s+\S+){8}\s+(\S+)', text, re.MULTILINE
    ):
        if not m.group(2).isdigit():
            non_est.append(m.group(1))
    return non_est


def parse_cisco_ping(text: str) -> dict:
    """Parse Cisco IOS ping: 'Success rate is 100 percent (5/5)'"""
    m = re.search(r'Success rate is (\d+) percent \((\d+)/(\d+)\)', text)
    if m:
        success_rate = int(m.group(1))
        return {"reachable": success_rate > 0, "loss_pct": 100 - success_rate}
    return {"reachable": False, "loss_pct": 100}
